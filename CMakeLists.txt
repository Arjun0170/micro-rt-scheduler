cmake_minimum_required(VERSION 3.16)
project(micro_rt_scheduler C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(firmware src/scheduler.c src/controller.c)
target_include_directories(firmware PUBLIC include)
target_compile_options(firmware PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(scheduler_sim sim/main.c)
target_link_libraries(scheduler_sim PRIVATE firmware)
add_executable(scheduler_tests tests/test_main.c)
target_link_libraries(scheduler_tests PRIVATE firmware)
enable_testing()
add_test(NAME scheduler_tests COMMAND scheduler_tests)
